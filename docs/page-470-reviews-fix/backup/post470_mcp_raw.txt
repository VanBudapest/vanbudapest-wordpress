  // Simple filter logic (client-side only)
  const list = document.getElementById('list');
  const q = document.getElementById('q');
  const country = document.getElementById('country');
  const year = document.getElementById('year');
  const count = document.getElementById('count');

  function refresh(){
    const term = (q.value||'').toLowerCase();
    const ctry = country.value;
    const yr = year.value;
    let visible = 0;
    list.querySelectorAll('.card').forEach(card =&gt;{
      const text = card.textContent.toLowerCase();
      const c = card.getAttribute('data-country');
      const d = card.getAttribute('data-date'); // yyyy-mm-dd
      const okTerm = !term || text.includes(term);
      const okCountry = !ctry || c === ctry;
      const okYear = !yr || (d &amp;&amp; d.startsWith(yr));
      const show = okTerm &amp;&amp; okCountry &amp;&amp; okYear;
      card.style.display = show ? '' : 'none';
      if(show) visible++;
    });
    count.textContent = visible + ' shown';
  }
  q.addEventListener('input', refresh);
  country.addEventListener('change', refresh);
  year.addEventListener('change', refresh);
  refresh();
